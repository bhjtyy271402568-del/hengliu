0163    if ((RCC->CFGR & RCC_CFGR_PPRE1) != 0)

0168    __HAL_TIM_DISABLE(&htim6);

0170    __HAL_TIM_SET_PRESCALER(&htim6, clk / 1000000 - 1);   // 1MHz
0171    __HAL_TIM_SET_AUTORELOAD(&htim6, 1000 - 1);           // 1ms

0266  // SHOOT 1ms读一次, 连续5ms一样才认
0267  // 光耦TLP2761是反相的, 外部有信号时PA8是低, 断线就是高
0268  static void Shoot_Debounce(void)
0269  {
0270    uint8_t now;
0271  
0272    if (!shoot_en)
0273      return;
0274  
0275    now = (HAL_GPIO_ReadPin(SHOOT_GPIO_Port, SHOOT_Pin) == GPIO_PIN_RESET) ? 1 : 0;
0276    if (now == shoot_active)
0277    {
0278      shoot_cnt = 0;
0279    }
0280    else if (++shoot_cnt >= SHOOT_DEBOUNCE_MS)
0281    {
0282      shoot_cnt = 0;
0283      shoot_active = now;
0284      shoot_on_req = now;   // 只有 无->有 才请求启动
0285    }
0286  
0287    // 信号撤了直接关. fault_cnt不为0是STOP在滤波, 等它判完
0288    if (!shoot_active && fault_cnt == 0 &&
0289        (laser_state == LASER_STARTING || laser_state == LASER_RUNNING))
0290    {
0291      Laser_Stop();
0292    }
0293  }

0457  // 出光时电流在目标值+-20%以内持续100ms, 指示灯亮
0458  static void Current_LED_Update(void)
0459  {
0460    uint32_t meas, target;
0461  
0462    if (ismon_age < CUR_TIMEOUT_MS)
0463      ismon_age++;
0464  
0465    /* 换算: CTRL = 30*Rs*I, ISMON = 20*Rs*I
0466     * ISMON经R2/R19(10k/8.2k)分压进ADC, ADC和DAC共用VREF+
0467     * 目标ADC = DAC * (8.2/18.2) * (20/30) = DAC * 82 / 273
0468     * TODO: 按电阻标称值算的, 没有实际标定过
0469     */
0470    target = (uint32_t)dac_target * 82;
0471    ismon_target = (uint16_t)((target + 136) / 273);
0472    meas = (uint32_t)ismon_adc * 273 * 100;
0473  
0474    if (laser_state == LASER_RUNNING && target > 0 && fault_cnt == 0 &&
0475        HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_SET &&
0476        ismon_ok && ismon_age < CUR_TIMEOUT_MS &&
0477        meas >= target * (100 - CUR_TOL_PCT) &&
0478        meas <= target * (100 + CUR_TOL_PCT))
0479    {
0480      if (ismon_good_cnt < CUR_OK_MS)
0481        ismon_good_cnt++;
0482    }
0483    else
0484    {
0485      ismon_good_cnt = 0;
0486    }
0487  
0488    // 故障时灯给Fault_LED_Blink用
0489    if (laser_state != LASER_FAULT)
0490    {
0491      HAL_GPIO_WritePin(R_EN_GPIO_Port, R_EN_Pin,
0492                        ismon_good_cnt >= CUR_OK_MS ? GPIO_PIN_SET : GPIO_PIN_RESET);
0493    }
0494  }
