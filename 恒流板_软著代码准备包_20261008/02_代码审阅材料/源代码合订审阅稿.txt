0212  // 启动屏蔽时间: DAC缓升要的时间再加100ms, 最多200ms
0213  static uint16_t Calc_Mask_Time(uint16_t dac)
0214  {
0215    uint32_t t = ((uint32_t)dac + DAC_STEP - 1) / DAC_STEP + START_MASK_EXTRA;
0216  
0217    if (t > START_MASK_MAX) t = START_MASK_MAX;
0218    return (uint16_t)t;
0219  }

0221  // 1ms一次, DAC每次加DAC_STEP, 加到目标值为止
0222  static void DAC_Ramp(void)
0223  {
0224    uint16_t next;
0225  
0226    if (laser_state != LASER_STARTING && laser_state != LASER_RUNNING)
0227      return;
0228    if (dac_now >= dac_target)
0229      return;
0230  
0231    next = dac_now + DAC_STEP;
0232    if (next < dac_now || next > dac_target)
0233    {
0234      next = dac_target;
0235    }
0236    dac_now = next;
0237    DAC_SetValue(next);
0238  }

0240  // TIM6里轮询STOP脚, 和STOP中断共用fault_cnt
0241  // 连续5次(5ms)都是低电平才锁故障
0242  static void Stop_Check(void)
0243  {
0244    if (laser_state != LASER_RUNNING)
0245    {
0246      return;
0247    }
0248  
0249    if (HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_RESET)
0250    {
0251      if (fault_cnt == 0)
0252      {
0253        Stop_IT_Off();
0254        fault_cnt = FAULT_FILTER_MS;
0255      }
0256      fault_cnt--;
0257      if (fault_cnt == 0)
0258      {
0259        Laser_Fault_Lock();
0260      }
0261    }
0262    else if (fault_cnt != 0)
0263    {
0264      // 低电平没持续够, 当干扰处理, 重新开中断
0265      fault_cnt = 0;
0266      if (Stop_IT_On() == 0)
0267      {
0268        // 开中断时又变低了, 重新计数
0269        fault_cnt = FAULT_FILTER_MS;
0270      }
0271    }
0272  }

0274  // SHOOT输入, TIM6里1ms读一次, 连续5次一样才认
0275  // 端子 -> R18 -> TLP2761 -> R55 -> PA8, 光耦是反相的:
0276  // 外部有信号 -> PA8低, 没信号/断线 -> PA8高(停)
0277  static void Shoot_Debounce(void)
0278  {
0279    uint8_t now;
0280  
0281    if (!shoot_en)
0282      return;
0283  
0284    now = (HAL_GPIO_ReadPin(SHOOT_GPIO_Port, SHOOT_Pin) == GPIO_PIN_RESET) ? 1 : 0;
0285    if (now == shoot_active)
0286    {
0287      shoot_cnt = 0;
0288    }
0289    else if (++shoot_cnt >= SHOOT_DEBOUNCE_MS)
0290    {
0291      shoot_cnt = 0;
0292      shoot_active = now;
0293      shoot_on_req = now;   // 只有 无->有 才请求启动
0294    }
0295  
0296    // 信号没了就在中断里直接关, 不等主循环
0297    // STOP正在滤波就等它判完再关, 不然fault_cnt会被Laser_Stop清掉
0298    if (!shoot_active && fault_cnt == 0 &&
0299        (laser_state == LASER_STARTING || laser_state == LASER_RUNNING))
0300    {
0301      Laser_Stop();
0302    }
0303  }

0468  // 出光时电流在目标值+-20%以内持续100ms, 指示灯亮
0469  static void Current_LED_Update(void)
0470  {
0471    uint32_t meas, target;
0472  
0473    if (ismon_age < CUR_TIMEOUT_MS)
0474      ismon_age++;
0475  
0476    /* 换算: CTRL = 30*Rs*I, ISMON = 20*Rs*I
0477     * ISMON经R2/R19(10k/8.2k)分压进ADC, ADC和DAC共用VREF+
0478     * 目标ADC = DAC * (8.2/18.2) * (20/30) = DAC * 82 / 273
0479     * TODO: 按电阻标称值算的, 没有实际标定过
0480     */
0481    target = (uint32_t)dac_target * 82;
0482    ismon_target = (uint16_t)((target + 136) / 273);
0483    meas = (uint32_t)ismon_adc * 273 * 100;
0484  
0485    if (laser_state == LASER_RUNNING && target > 0 && fault_cnt == 0 &&
0486        HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_SET &&
0487        ismon_ok && ismon_age < CUR_TIMEOUT_MS &&
0488        meas >= target * (100 - CUR_TOL_PCT) &&
0489        meas <= target * (100 + CUR_TOL_PCT))
0490    {
0491      if (ismon_good_cnt < CUR_OK_MS)
0492        ismon_good_cnt++;
0493    }
0494    else
0495    {
0496      ismon_good_cnt = 0;
0497    }
0498  
0499    // 故障时灯给Fault_LED_Blink用
0500    if (laser_state != LASER_FAULT)
0501    {
0502      HAL_GPIO_WritePin(R_EN_GPIO_Port, R_EN_Pin,
0503                        ismon_good_cnt >= CUR_OK_MS ? GPIO_PIN_SET : GPIO_PIN_RESET);
0504    }
0505  }

0507  // 故障锁存后指示灯250ms翻转一次
0508  static void Fault_LED_Blink(void)
0509  {
0510    if (laser_state != LASER_FAULT)
0511      return;
0512  
0513    if (led_blink_cnt > 0)
0514      led_blink_cnt--;
0515

0517    {
0518      HAL_GPIO_TogglePin(R_EN_GPIO_Port, R_EN_Pin);
0519      led_blink_cnt = FAULT_BLINK_MS;
0520    }
0521  }
