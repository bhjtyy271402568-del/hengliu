0188  static void Laser_Process_Current_LED(void)
0189  {
0190    uint32_t measured;
0191    uint32_t target;
0192  
0193    if (g_laser_current_age_ms < LASER_CURRENT_STALE_MS)
0194    {
0195      g_laser_current_age_ms++;
0196    }
0197  
0198    /* R2/R19 = 10k/8.2k, CTRL = 30*Rs*I, ISMON = 20*Rs*I.
0199       ADC and DAC share VREF+: target ADC = DAC * (41/91) * (20/30).
0200       This nominal ratio is NOT an ISMON accuracy calibration. */
0201    target = (uint32_t)g_laser_dac_target_val * 82U;
0202    g_laser_target_adc = (uint16_t)((target + 136U) / 273U);
0203    measured = (uint32_t)g_laser_current_adc * 273U * 100U;
0204  
0205    if ((g_laser_state == LASER_STATE_RUNNING) && (target > 0U) &&
0206        (g_laser_fault_debounce_ms == 0U) &&
0207        (HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_SET) &&
0208        (g_laser_current_valid != 0U) &&
0209        (g_laser_current_age_ms < LASER_CURRENT_STALE_MS) &&
0210        (measured >= target * (100U - LASER_CURRENT_TOLERANCE_PERCENT)) &&
0211        (measured <= target * (100U + LASER_CURRENT_TOLERANCE_PERCENT)))
0212    {
0213      if (g_laser_current_good_ms < LASER_CURRENT_GOOD_MS)
0214      {
0215        g_laser_current_good_ms++;
0216      }
0217    }
0218    else
0219    {
0220      g_laser_current_good_ms = 0U;
0221    }
0222  
0223    /* Fault blinking owns the LED while FAULT is latched. */
0224    if (g_laser_state != LASER_STATE_FAULT)
0225    {
0226      HAL_GPIO_WritePin(R_EN_GPIO_Port, R_EN_Pin,
0227                       (g_laser_current_good_ms >= LASER_CURRENT_GOOD_MS) ? GPIO_PIN_SET : GPIO_PIN_RESET);
0228    }
0229  }

0252  static void Laser_Process_Stop_Fault(void)
0253  {
0254    if (g_laser_state != LASER_STATE_RUNNING)
0255    {
0256      return;
0257    }
0258  
0259    if (HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_RESET)
0260    {
0261      if (g_laser_fault_debounce_ms == 0U)
0262      {
0263        Laser_Disable_Stop_EXTI();
0264        g_laser_fault_debounce_ms = LASER_FAULT_DEBOUNCE_MS;
0265      }
0266  
0267      /* EXTI and polling share five consecutive 1 ms low samples. */
0268      g_laser_fault_debounce_ms--;
0269      if (g_laser_fault_debounce_ms == 0U)
0270      {
0271        Laser_Emergency_Kill();
0272      }
0273    }
0274    else if (g_laser_fault_debounce_ms != 0U)
0275    {
0276      g_laser_fault_debounce_ms = 0U;
0277      if (Laser_Enable_Stop_EXTI() == 0U)
0278      {
0279        /* A new falling edge during re-arm needs its own debounce. */
0280        g_laser_fault_debounce_ms = LASER_FAULT_DEBOUNCE_MS;
0281      }
0282    }
0283  }

0529    uint32_t mask_ms = ramp_ms + LASER_START_MASK_EXTRA_MS;

0534    }

0536    return (uint16_t)mask_ms;

0539  static void Laser_Process_Dac_Ramp(void)
0540  {
0541    uint16_t next_val;
0542  
0543    if ((g_laser_state != LASER_STATE_STARTING) && (g_laser_state != LASER_STATE_RUNNING))
0544    {
0545      return;
0546    }
0547  
0548    if (g_laser_dac_current_val >= g_laser_dac_target_val)
0549    {
0550      return;
0551    }
0552  
0553    next_val = g_laser_dac_current_val + LASER_DAC_RAMP_STEP_PER_MS;
0554    if ((next_val < g_laser_dac_current_val) || (next_val > g_laser_dac_target_val))
0555    {
0556      next_val = g_laser_dac_target_val;
0557    }
0558  
0559    g_laser_dac_current_val = next_val;
0560    Laser_Set_Dac_12bit(next_val);
0561  }
