0224  // 1ms一次, DAC每次加DAC_STEP, 加到目标值为止
0225  static void DAC_Ramp(void)
0226  {
0227    uint16_t next;
0228  
0229    if (laser_state != LASER_STARTING && laser_state != LASER_RUNNING)
0230      return;
0231    if (dac_now >= dac_target)
0232      return;
0233  
0234    next = dac_now + DAC_STEP;
0235    if (next < dac_now || next > dac_target)
0236    {
0237      next = dac_target;
0238    }
0239    dac_now = next;
0240    DAC_SetValue(next);
0241  }

0243  // TIM6里轮询STOP脚, 和STOP中断共用fault_cnt
0244  // 连续5次(5ms)都是低电平才锁故障
0245  static void Stop_Check(void)
0246  {
0247    if (laser_state != LASER_RUNNING)
0248    {
0249      return;
0250    }
0251  
0252    if (HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_RESET)
0253    {
0254      if (fault_cnt == 0)
0255      {
0256        Stop_IT_Off();
0257        fault_cnt = FAULT_FILTER_MS;
0258      }
0259      fault_cnt--;
0260      if (fault_cnt == 0)
0261      {
0262        Laser_Fault_Lock();
0263      }
0264    }
0265    else if (fault_cnt != 0)
0266    {
0267      // 低电平没持续够, 当干扰处理, 重新开中断
0268      fault_cnt = 0;
0269      if (Stop_IT_On() == 0)
0270      {
0271        // 开中断时又变低了, 重新计数
0272        fault_cnt = FAULT_FILTER_MS;
0273      }
0274    }
0275  }

0299      {

0472  // 出光时电流在目标值+-20%以内持续100ms, 指示灯亮
0473  static void Current_LED_Update(void)
0474  {
0475    uint32_t meas, target;
0476  
0477    if (ismon_age < CUR_TIMEOUT_MS)
0478      ismon_age++;
0479  
0480    /* 换算: CTRL = 30*Rs*I, ISMON = 20*Rs*I
0481     * ISMON经R2/R19(10k/8.2k)分压进ADC, ADC和DAC共用VREF+
0482     * 目标ADC = DAC * (8.2/18.2) * (20/30) = DAC * 82 / 273
0483     * TODO: 按电阻标称值算的, 没有实际标定过
0484     */
0485    target = (uint32_t)dac_target * 82;
0486    ismon_target = (uint16_t)((target + 136) / 273);
0487    meas = (uint32_t)ismon_adc * 273 * 100;
0488  
0489    if (laser_state == LASER_RUNNING && target > 0 && fault_cnt == 0 &&
0490        HAL_GPIO_ReadPin(STOP_GPIO_Port, STOP_Pin) == GPIO_PIN_SET &&
0491        ismon_ok && ismon_age < CUR_TIMEOUT_MS &&
0492        meas >= target * (100 - CUR_TOL_PCT) &&
0493        meas <= target * (100 + CUR_TOL_PCT))
0494    {
0495      if (ismon_good_cnt < CUR_OK_MS)
0496        ismon_good_cnt++;
0497    }
0498    else
0499    {
0500      ismon_good_cnt = 0;
0501    }
0502  
0503    // 故障时灯给Fault_LED_Blink用
0504    if (laser_state != LASER_FAULT)
0505    {
0506      HAL_GPIO_WritePin(R_EN_GPIO_Port, R_EN_Pin,
0507                        ismon_good_cnt >= CUR_OK_MS ? GPIO_PIN_SET : GPIO_PIN_RESET);
0508    }
0509  }
